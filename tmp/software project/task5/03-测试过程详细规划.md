  const startupMethods = [
    { cmd: 'node', args: ['server.js'], port: 3000 },
    { cmd: 'node', args: ['server.js', '3001'], port: 3001 },
    { cmd: 'node', args: ['server.js', 'ngrok'], port: 3000 },

  ];

      // 等待服务器启动
      await new Promise(resolve => setTimeout(resolve, 3000));
      
      // 检查健康检查端点
      const response = await fetch(`http://localhost:${port}/health`);
      expect(response.status).toBe(200);
      
      // 停止服务器
      server.kill();
    });
  });
});

### 4.1 Bug案例1：历史记录加载问题

#### 4.1.1 问题描述
**发现时间**: 2025年11月
**问题**: 历史消息加载异常，投票数据缺失

#### 4.1.2 PIE模型分析
- **Execution（执行）**: 异步循环执行过程中发生错误
- **Infection（感染）**: 错误导致投票数据未正确加载到消息对象中
- **Propagation（传播）**: 错误传播到前端，导致投票消息显示异常

#### 4.1.3 调试流程
1. **复现（Reproduce）**: 重现bug场景（加载包含投票的历史消息）
2. **定位（Locate）**: 确定bug的位置（joinRoom.js第124-141行的循环加载逻辑）
3. **理解（Understand）**: 理解bug的原因（异步循环错误处理不足，数据库查询失败未捕获）
4. **修复（Fix）**: 使用Promise.all并行加载，添加try-catch错误处理
5. **验证（Verify）**: 验证修复是否有效（重新加载历史消息，投票数据正常显示）

#### 4.1.4 修复代码
**原代码问题**:
```javascript
// 原代码：异步循环错误处理不足
for (let i = 0; i < history.length; i++) {
  const message = history[i];
  const poll = await db.getPollByMessageId(message.id);
  if (poll) {
    // 如果数据库查询失败，错误未被捕获
    const userVote = await db.getUserVote({
      pollId: poll.id,
      sessionId: socket.id
    });
    history[i] = { ...message, poll: { ...poll, userVote } };
  }
}
```

**修复后代码**:
```javascript
// 修复后：使用Promise.all并行加载，添加错误处理
const pollPromises = history.map(async (message) => {
  try {
    const poll = await db.getPollByMessageId(message.id);
    if (poll) {
      const userVote = await db.getUserVote({
        pollId: poll.id,
        sessionId: socket.id
      });
      return { ...message, poll: { ...poll, userVote } };
    }
    return message;
  } catch (err) {
    console.error(`[历史消息] 加载投票数据失败 (messageId: ${message.id}):`, err);
    return message; // 即使投票数据加载失败，也返回消息本身
  }
});
history = await Promise.all(pollPromises);
```

**修复要点**:
1. 使用Promise.all并行加载，提升性能
2. 添加try-catch错误处理，防止单个投票数据加载失败影响整体
3. 保持消息顺序
4. 即使投票数据加载失败，也返回消息本身，保证基本功能正常
